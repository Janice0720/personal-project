- [ ] **Step 2：生成 3,000 筆合成個體資料**

```python
np.random.seed(42)
N = 3000
records = []

groups = list(INCOME_PROFILES.keys())
# 依台灣實際申報戶比例分配（各 20%）
group_sizes = [600, 600, 600, 600, 600]

for group, n in zip(groups, group_sizes):
    profile = INCOME_PROFILES[group]
    base_income = INCOME_BASE_KTW[group]

    for _ in range(n):
        # 總所得加入隨機擾動（log-normal 分佈模擬真實所得分散程度）
        total_income = np.random.lognormal(
            mean=np.log(base_income * 1000),
            sigma=0.4
        )
        # 各類所得依比例分配，並加入 Dirichlet 擾動模擬個體差異
        proportions = np.array(list(profile.values()))
        proportions = proportions / proportions.sum()
        noisy_prop = np.random.dirichlet(proportions * 10)  # concentration=10 控制分散程度

        income_types = list(profile.keys())
        row = {'group': group, 'total_income': round(total_income)}
        for col, prop in zip(income_types, noisy_prop):
            row[col] = round(total_income * prop)

        # 模擬扣繳稅額（扣繳率約 5-20%，依所得高低）
        base_tax_rate = 0.05 + (groups.index(group) * 0.03)
        row['withholding_tax'] = round(total_income * np.random.uniform(base_tax_rate * 0.8, base_tax_rate * 1.2))
        records.append(row)

df_synthetic = pd.DataFrame(records)
print(df_synthetic.shape)
print(df_synthetic.describe())
```

## Task 3：特徵工程

```python
# scripts/feature_engineering.py
import numpy as np
import pandas as pd

INCOME_COLS = ['salary', 'dividend', 'interest', 'rental', 'business', 'other']


def compute_income_ratios(df: pd.DataFrame) -> pd.DataFrame:
    """計算各類所得佔總所得的比例（0~1），作為 K-Means 特徵"""
    df = df.copy()
    for col in INCOME_COLS:
        df[f'{col}_ratio'] = df[col] / df['total_income'].replace(0, np.nan)
    df = df.fillna(0)
    return df


def compute_shannon_entropy(df: pd.DataFrame) -> pd.Series:
    """
    計算所得多元化程度（Shannon Entropy）。
    值越高代表所得來源越分散、越穩健。
    公式：H = -sum(p * log2(p))，p 為各類所得佔比
    """
    ratio_cols = [f'{col}_ratio' for col in INCOME_COLS]
    ratios = df[ratio_cols].values
    # 避免 log(0)：將 0 替換為極小值
    ratios_safe = np.where(ratios > 0, ratios, 1e-10)
    entropy = -np.sum(ratios_safe * np.log2(ratios_safe), axis=1)
    return pd.Series(entropy, index=df.index, name='shannon_entropy')


def compute_tax_burden_rate(df: pd.DataFrame) -> pd.Series:
    """
    計算稅務負擔率 = 扣繳稅額 / 總所得
    用於稅務效率評分（對照同分位平均值）
    """
    rate = df['withholding_tax'] / df['total_income'].replace(0, np.nan)
    return rate.fillna(0).rename('tax_burden_rate')


def build_features(df: pd.DataFrame) -> pd.DataFrame:
    """整合所有特徵，回傳可供 K-Means 使用的特徵 DataFrame"""
    df = compute_income_ratios(df)
    df['shannon_entropy'] = compute_shannon_entropy(df)
    df['tax_burden_rate'] = compute_tax_burden_rate(df)
    return df
```

```python
# scripts/clustering.py
import joblib
import numpy as np
import pandas as pd
from sklearn.cluster import KMeans
from sklearn.preprocessing import StandardScaler

FEATURE_COLS = ['salary_ratio', 'dividend_ratio', 'interest_ratio',
                'rental_ratio', 'business_ratio', 'shannon_entropy']

CLUSTER_NAMES = {
    0: '薪資型',
    1: '投資型',
    2: '自雇型',
    3: '混合型',
}


def train_and_save(df: pd.DataFrame, k: int = 4, model_path: str = '../models/kmeans_model.pkl'):
    X = df[FEATURE_COLS].values
    scaler = StandardScaler()
    X_scaled = scaler.fit_transform(X)
    km = KMeans(n_clusters=k, random_state=42, n_init=10)
    km.fit(X_scaled)
    joblib.dump({'model': km, 'scaler': scaler, 'feature_cols': FEATURE_COLS}, model_path)
    print(f"模型已儲存至 {model_path}")
    return km, scaler


def predict_persona(user_features: dict, model_path: str = '../models/kmeans_model.pkl') -> str:
    """
    輸入使用者所得特徵字典，回傳財務人物類型。
    user_features 範例：
    {'salary_ratio': 0.8, 'dividend_ratio': 0.05, 'interest_ratio': 0.1,
     'rental_ratio': 0.0, 'business_ratio': 0.05, 'shannon_entropy': 0.9}
    """
    bundle = joblib.load(model_path)
    km, scaler, cols = bundle['model'], bundle['scaler'], bundle['feature_cols']
    X = np.array([[user_features[c] for c in cols]])
    X_scaled = scaler.transform(X)
    cluster = km.predict(X_scaled)[0]
    return CLUSTER_NAMES.get(cluster, '未知類型')
```

```python
# scripts/scoring.py
import numpy as np
import pandas as pd

# Shannon Entropy 最大值（6 類所得均等時）= log2(6) ≈ 2.585
MAX_ENTROPY = np.log2(6)


def score_diversification(shannon_entropy: float) -> int:
    """
    所得多元化評分（0～100）。
    Shannon Entropy 越高 → 評分越高。
    """
    score = (shannon_entropy / MAX_ENTROPY) * 100
    return int(np.clip(score, 0, 100))


def score_tax_efficiency(user_tax_rate: float, benchmark_tax_rate: float) -> int:
    """
    稅務效率評分（0～100）。
    user_tax_rate: 使用者扣繳稅率（扣繳稅額 / 總所得）
    benchmark_tax_rate: 同所得分位的平均扣繳稅率
    比使用者稅率低 → 代表稅務效率高 → 高分
    比使用者稅率高 → 代表可能有節稅空間 → 低分
    """
    diff = benchmark_tax_rate - user_tax_rate
    # diff > 0 代表使用者稅率低於基準（好），diff < 0 代表高於基準（差）
    score = 50 + (diff / benchmark_tax_rate) * 50
    return int(np.clip(score, 0, 100))


# 各分位的基準扣繳稅率（從合成資料計算，或從財政部統計推算）
BENCHMARK_TAX_RATES = {
    'low':     0.05,
    'mid_low': 0.08,
    'mid':     0.11,
    'mid_high':0.14,
    'high':    0.17,
}


def get_fincheck_score(shannon_entropy: float,
                       user_tax_rate: float,
                       income_group: str) -> dict:
    """
    整合健檢評分，回傳各維度分數。
    """
    bench = BENCHMARK_TAX_RATES.get(income_group, 0.10)
    return {
        'diversification_score': score_diversification(shannon_entropy),
        'tax_efficiency_score':  score_tax_efficiency(user_tax_rate, bench),
    }
```

```python
        tips.append("所得多元化程度尚可，可考慮進一步分散至不同資產類型（如債券基金、REITs）。")
    else:
        tips.append("所得來源多元，財務結構穩健，建議定期檢視各資產類別比例是否符合人生階段規劃。")

    if tax_efficiency_score < 50:
        tips.append("你的稅務負擔率高於同所得層平均，建議諮詢理財顧問，檢視是否善用合法扣除額（如捐贈、醫療支出）。")
    else:
        tips.append("稅務效率表現良好，持續保持，並留意每年 5 月申報期可申請退稅的項目。")

    persona_tips = {
        '薪資型': "薪資佔所得比例高，建議優先建立 3～6 個月緊急預備金後，再開始長期投資計畫。",
        '投資型': "股利與利息所得佔比高，建議定期檢視持股集中度，避免單一標的風險過高。",
        '自雇型': "執行業務所得屬浮動性收入，建議強化退休規劃（如勞退自提），並考慮保障型保險。",
        '混合型': "所得結構均衡，建議進一步優化稅務規劃，充分利用各類所得的扣除優惠。",
    }
    tips.append(persona_tips.get(persona, "建議定期進行財務健檢，根據人生階段調整資產配置策略。"))

    return tips[:3]  # 最多回傳 3 條
```

```python
if st.button("開始健檢", type="primary"):
    total = salary + dividend + interest + rental + business
    if total == 0:
        st.error("請至少輸入一項所得金額")
        st.stop()

    # 計算所得佔比
    ratios = {
        'salary_ratio':   salary   / total,
        'dividend_ratio': dividend / total,
        'interest_ratio': interest / total,
        'rental_ratio':   rental   / total,
        'business_ratio': business / total,
        'other_ratio':    0.0,
    }

    # Shannon Entropy
    vals = np.array(list(ratios.values()))
    vals_safe = np.where(vals > 0, vals, 1e-10)
    entropy = -np.sum(vals_safe * np.log2(vals_safe))
    ratios['shannon_entropy'] = entropy

    # 判斷所得分位
    if total < 1_000_000:
        group = 'low'
    elif total < 1_300_000:
        group = 'mid_low'
    elif total < 1_600_000:
        group = 'mid'
    elif total < 2_000_000:
        group = 'mid_high'
    else:
        group = 'high'

    tax_rate = withholding_tax / total
    persona = predict_persona(ratios)
    scores = get_fincheck_score(entropy, tax_rate, group)
    tips = generate_recommendations(persona, scores['diversification_score'], scores['tax_efficiency_score'])

    # --- 顯示結果 ---
    st.divider()
    st.header("Step 2｜你的財務健檢報告")

    col_a, col_b = st.columns([1, 2])
    with col_a:
        st.subheader("財務人物類型")
        st.metric("你是", persona)
        st.metric("所得多元化評分", f"{scores['diversification_score']} 分")
        st.metric("稅務效率評分",   f"{scores['tax_efficiency_score']} 分")

    with col_b:
        # 所得結構圓餅圖
        labels = ['薪資', '股利', '利息', '租賃', '執行業務']
        values = [salary, dividend, interest, rental, business]
        fig_pie = px.pie(values=values, names=labels, title="你的所得結構分佈")
        st.plotly_chart(fig_pie, use_container_width=True)

    # 評分雷達圖
    fig_radar = go.Figure(go.Scatterpolar(
        r=[scores['diversification_score'], scores['tax_efficiency_score'], 50],
        theta=['所得多元化', '稅務效率', '整體健康度'],
        fill='toself'
    ))
    fig_radar.update_layout(polar=dict(radialaxis=dict(range=[0, 100])),
                            title="財務健檢雷達圖")
    st.plotly_chart(fig_radar, use_container_width=True)

    # AI 建議
    st.subheader("個人化理財建議")
    for i, tip in enumerate(tips, 1):
        st.info(f"{i}. {tip}")
```
